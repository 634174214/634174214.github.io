PHP接口数据加密实战

先理清：3个核心概念，避免用错场景

很多人混淆“加密”“哈希”“签名”，用错场景反而增加安全隐患，先花1分钟搞懂，避免踩坑：

  • 加密：可逆操作，把明文（比如手机号13800138000）变成密文，拿到密钥就能解密回明文，适合敏感数据传输（如手机号、身份证）；

  • 哈希：不可逆操作，把数据变成固定长度的字符串（比如密码加密），无法还原回原文，适合密码存储、数据完整性校验；

  • 签名：结合哈希+密钥，验证数据是否被篡改，发送方生成签名，接收方验证签名，不一致则拒绝处理，适合接口请求防篡改。

记住一句话：敏感数据用加密，密码存储用哈希，接口防篡改用签名，三者结合，才能真正保障接口数据安全。

实战方案1：AES对称加密（前后端数据传输首选）

AES是目前最常用的对称加密算法，加密和解密用同一个密钥，速度快、兼容性好，适合前后端敏感数据传输（如手机号、地址、订单信息），PHP原生支持，不用额外安装扩展。

步骤1：封装AES加密工具类（直接复制用）

<?php
/** 
* AES加密工具类（AES-256-CBC，生产环境首选） 
* 前后端需统一：密钥、加密算法、编码方式
*/
class AesEncrypt{
	    
	  // 密钥（重点：前后端必须一致，建议32位，可自定义，不要泄露）    private static $key = 'your_32_byte_aes_secret_key_2026';    
	// 加密算法（固定AES-256-CBC，兼容性最好）    
	private static $cipher = 'aes-256-cbc';
	    
	/**     
  * AES加密（明文转密文，方便传输）     
  * @param string $data 待加密明文（可传字符串、数组）     
  * @return string 加密后的密文（base64编码，避免乱码）     
  */
	    
	  public static function encrypt($data)    
	  {
		        
		  // 处理数组：先JSON编码，再加密        
		if (is_array($data)) {
			            
			  $data = json_encode($data, JSON_UNESCAPED_UNICODE);
			       
		}
		        
		  // 生成初始化向量IV（16位，AES-256-CBC固定要求）        $iv = openssl_random_pseudo_bytes(openssl_cipher_iv_length(self::$cipher));        
		// 执行加密        
		$encrypted = openssl_encrypt($data, self::$cipher, self::$key, OPENSSL_RAW_DATA, $iv);
		        
		  // 拼接IV和密文，base64编码（方便传输，避免乱码）        return base64_encode($iv . $encrypted);   
	}
	   
	/**     
  * AES解密（密文转明文）     
  * @param string $encryptedData 加密后的密文（base64编码）     
  * @return string|array 解密后的明文（数组会自动JSON解码）     
  * @throws Exception 解密失败抛出异常     
  */
	    
	  public static function decrypt($encryptedData)    
	  {
		        
		  // 先base64解码        
		$decoded = base64_decode($encryptedData);
		        
		  if ($decoded === false) {
			            
			  throw new Exception("加密数据格式错误，无法解码");
			       
		}
		        
		  // 拆分IV和密文（IV固定16位）        
		$ivLength = openssl_cipher_iv_length(self::$cipher);
		        $iv = substr($decoded, 0, $ivLength);
		        
		  $encrypted = substr($decoded, $ivLength);
		        
		  // 执行解密        
		$decrypted = openssl_decrypt($encrypted, self::$cipher, self::$key,OPENSSL_RAW_DATA, $iv);
		       
		  if ($decrypted === false) {
			           
			  throw new Exception("解密失败，密钥或数据错误");
			       
		}
		        
		  // 尝试JSON解码（如果是数组，自动还原）        
		$jsonDecoded = json_decode($decrypted, true);
		        return $jsonDecoded ?? $decrypted;
		   
	}
}
?>

步骤2：接口中实战使用（前后端对接示例）

后端加密返回数据，前端解密展示；前端加密请求参数，后端解密处理，完美实现敏感数据隐藏。

<?php
// 示例1：后端加密返回敏感数据（比如用户信息）
public function getUserInfo()
{
	// 模拟用户敏感数据    
	$userInfo = [        
	'id' => 1001,       
	'name' => '张三',        
	'phone' => '13800138000', 
	// 敏感数据，需要加密       
	'address' => '北京市朝阳区' 
	// 敏感数据，需要加密    
	];
	try {
		// 加密敏感数据       
		$encryptedUserInfo = AesEncrypt::encrypt($userInfo);
		// 接口返回密文        
		apiResponse(
		0, 
		'success', 
		[           
		'user_info' => $encryptedUserInfo        
		]);
	}
	catch (Exception $e) {
		apiResponse(500, $e->getMessage());
	}
}
// 示例2：后端解密前端加密的请求参数（比如提交订单）
public function submitOrder(){
	// 接收前端加密后的参数    
	$encryptedParams = $_POST['encrypted_params'] ?? '';
	if (empty($encryptedParams)) {
		apiResponse(400, '请提交加密参数');
	}
	try {
		// 解密参数        
		$params = AesEncrypt::decrypt($encryptedParams);
		// 解密后执行业务逻辑（比如订单处理）        
		$orderId = $this->createOrder($params['goods_id'], $params['amount']);
		apiResponse(0, '订单提交成功', ['order_id' => $orderId]);
	}
	catch (Exception $e) {
		apiResponse(400, $e->getMessage());
	}
}
?>

前端配合示例（简单易懂，适配Vue/React）

前端用相同的密钥和算法，实现加密解密，和后端保持一致即可（以JS为例）

// 前端AES加密解密（和PHP后端保持一致
const key = 'your_32_byte_aes_secret_key_2026';
// 和后端密钥一致
const cipher = 'AES-256-CBC';
// 加密函数（和PHP逻辑对应）
function aesEncrypt(data) {
	const dataStr = typeof data === 'object' ? JSON.stringify(data) : data;
	const iv = crypto.getRandomValues(new Uint8Array(16));
	// 16位IV    
	const encrypted = crypto.subtle.encrypt(       
	{ name: cipher, iv: iv },        key,        new TextEncoder().encode(dataStr)    );
	// 拼接IV和密文，base64编码   
	return btoa(String.fromCharCode(...new Uint8Array(await encrypted)));
}
// 解密函数async function aesDecrypt(encryptedData) 
{
	const decoded = atob(encryptedData);
	const iv = new Uint8Array(Array.from(decoded.slice(0, 16)).map(char => char.charCodeAt(0)));
	const encrypted = new Uint8Array(Array.from(decoded.slice(16)).map(char => char.charCodeAt(0)));
	const decrypted = await crypto.subtle.decrypt(      
	{ name: cipher, iv: iv },       
	key,        
	encrypted   
	);
	const dataStr = new TextDecoder().decode(decrypted);
	try {
		return JSON.parse(dataStr);
		// 尝试解析为数组
	}
	catch (e) {
		return dataStr;
	}
}
// 调用示例：加密请求参数提交
const params = { goods_id: 101, amount: 199 };
const encryptedParams = await aesEncrypt(params);
// 发起请求，提交加密后的参数axios.post('/api/submitOrder', 
{
	encrypted_params: encryptedParams
}
);

实战方案2：HMAC签名（接口防篡改首选）

有些场景不需要加密数据（比如公开列表），但需要防止参数被篡改（比如订单金额、用户ID），这时候就用HMAC签名——发送方生成签名，接收方验证签名，只要参数被篡改，签名就会不一致，直接拒绝请求。

核心优势：不用加密数据，只验证完整性，速度快，适合高频接口、第三方对接。

步骤1：封装签名工具类（直接复制用）

<?php
/** * HMAC签名工具类（防篡改，支持GET/POST请求） 
* 前后端需统一：签名密钥、参数排序规则、签名算法 *
/class HmacSign{    
// 签名密钥（重点：前后端必须一致，不要泄露，和AES密钥可不同）   
private static $secret = 'your_hmac_secret_key_2026';   
// 签名算法（固定SHA256，安全性高）    
private static $algorithm = 'sha256'; 
/**     
* 生成签名    
* @param array $params 待签名的请求参数（不含sign字段）    
* @return string 生成的签名     
*/
public static function createSign($params)    
{
	// 1. 移除sign字段（避免重复签名）        
	unset($params['sign']);
	// 2. 按参数名ASCII排序（前后端必须一致，否则签名不一致）        
	ksort($params);
	// 3. 拼接参数为字符串（key=value&key=value）      
	$paramStr = http_build_query($params);
	// 4. 生成HMAC签名（密钥+参数字符串，SHA256算法）      
	return hash_hmac(self::$algorithm, $paramStr, self::$secret);
}
/**     
* 验证签名     
* @param array $params 接收的请求参数（含sign字段）     
* @return bool 签名是否有效     
*/
public static function verifySign($params)   
{
	// 1. 判断sign字段是否存在        
	if (!isset($params['sign']) || empty($params['sign'])) {
		return false;
	}
	// 2. 保存接收的签名，然后移除sign字段    
	$receivedSign = $params['sign'];
	unset($params['sign']);
	// 3. 重新生成签名，和接收的签名对比      
	$generatedSign = self::createSign($params);
	return $generatedSign === $receivedSign;
}
}
?>

步骤2：接口中实战使用（防篡改示例）

<?php
// 示例：订单支付接口（防止金额被篡改）
public function payOrder()
{
	// 接收请求参数（含sign签名）   
	$params = $_POST;
	// 1. 验证签名    
	if (!HmacSign::verifySign($params)) 
	{
		apiResponse(403, '参数被篡改，请求无效');
	}
	// 2. 签名验证通过，执行业务逻辑    
	$orderId = $params['order_id'] ?? '';
	$amount = $params['amount'] ?? 0;
	// 金额不会被篡改，可安全使用    
	// 模拟支付逻辑   
	$payResult = $this->doPay($orderId, $amount);
	if ($payResult) {
		apiResponse(0, '支付成功');
	} else {
		apiResponse(400, '支付失败，请重试');
	}
}
// 前端请求示例（生成签名）
// 前端拼接参数：
$params = ['order_id' => 10001, 'amount' => 199, 'timestamp' => time()];
// 生成签名：$sign = HmacSign::createSign($params);
// 提交参数：$params['sign'] = $sign; 然后发起POST请求
?>

生产环境必加的4个安全细节（避坑关键）

  1. 密钥绝对不能泄露：AES密钥、HMAC签名密钥，不要写死在代码里提交到GitHub，生产环境建议放在配置文件（如.env），权限设置为只读，避免被恶意获取；

  2. 前后端配置必须完全一致：加密算法、密钥、参数排序规则、编码方式，只要有一点不一样，就会出现“加密成功、解密失败”“签名验证不通过”的问题，对接前一定要和前端确认；

  3. 敏感数据必须加密：手机号、身份证、银行卡号、地址等敏感信息，必须用AES加密传输，禁止明文传输，即使有HTTPS，也建议加密（双重保障）；

  4. 签名加时间戳防重放：在签名参数中加入timestamp（时间戳），验证时判断时间戳是否在有效范围内（比如±5分钟），防止恶意重复提交请求（比如重复支付）。

接口数据安全，核心就是“加密+签名”，按场景选择即可：

敏感数据传输（手机号、身份证）：用AES对称加密，可逆、速度快，前后端统一密钥；

接口防篡改（订单金额、用户ID）：用HMAC签名，不用加密数据，只验证完整性；

密码存储：用PHP原生的password_hash()（哈希，不可逆），不要用AES加密密码（可逆，不安全）。

把今天的两个工具类复制到你的项目里，按需修改密钥，就能快速实现接口数据加密和防篡改，让你的接口更安全、更可靠。

接口安全没有“差不多”，只有“万无一失”——很多时候，一次数据泄露、一次参数篡改，就能造成无法挽回的损失。加密和签名，就是最基础、最有效的防护手段。
